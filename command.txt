
## Установка Docker (если нет)

### 
sudo apt update
sudo apt install -y docker.io docker-compose-v2
sudo usermod -aG docker $USER
newgrp docker





# 1. Разархивировать
tar -xzf ecmwf-minio-loader.tar.gz
cd ecmwf-minio-loader

# 2. Запустить
docker compose up -d

# 3. Проверить логи
docker compose logs -f ecmwf-loader





Конфигурация
Основные параметры (src/config.py)
max_parallel_downloads	20	Количество одновременных загрузок
retry_count	5	Повторных попыток при ошибке



1. Постоянный сервис (основной)
bash
docker compose up -d
Проверяет появление данных 00z каждые 10 минут с 6:00 до 12:00

Автоматически загружает при появлении

Каждый час проверяет, не пропущены ли данные (12:00-18:00)

2. Разовая загрузка
bash
docker compose run --rm ecmwf-loader python3 main.py --once --date 20260803

3. Локальный запуск (без Docker)
bash
pip install -r requirements.txt
# Изменить MINIO_ENDPOINT в .env на localhost:10000
python3 main.py --once --date 20260803


Доступ к Minio
Minio API	http://localhost:10000	minioadmin	adminminio



Управление
# Статус контейнеров
docker compose ps

# Логи
docker compose logs -f ecmwf-loader

# Остановка
docker compose down

# Остановка с удалением данных Minio
docker compose down -v

# Пересборка после изменений кода
docker compose build ecmwf-loader && docker compose up -d
